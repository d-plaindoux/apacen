  type_system(
        Strategy,
        Gamma |- (A @ B) : T1,
        proof(abstraction,LOG1,LOG2,RED)
    ) :-
    member(Strategy,check::infer_type::nil),
    !, -- green cut
    type_system(Gamma |- A : (X:R) -> T2,LOG1),
    type_system(Gamma |- B : R,LOG2),
    beta(Gamma,T2[X:=B],T1,RED).
